CREATE DATABASE Belle_Croissant_Lyonnais;

USE Belle_Croissant_Lyonnais;

CREATE TABLE Pregunta(
Pregunta_ID INT PRIMARY KEY IDENTITY(1,1) NOT NULL,
Pregunta VARCHAR(50) NOT NULL
);

CREATE TABLE Usuario(
Usuario_ID INT PRIMARY KEY IDENTITY(1,1) NOT NULL,
Pregunta_ID INT NOT NULL,
Nombre VARCHAR(50) NOT NULL,
Apellido VARCHAR(50) NOT NULL,
Email VARCHAR(100) NOT NULL UNIQUE,
Password VARCHAR(MAX) NOT NULL,
Telefono VARCHAR(20),
Foto_Perfil VARCHAR(150),
Suscripcion BIT,
Metodo_Entrega BIT,
Respuesta_Segurtidad VARCHAR(150) NOT NULL,

FOREIGN KEY (Pregunta_ID) REFERENCES Pregunta(Pregunta_ID)
);

CREATE TABLE Direccion(
Direccion_ID INT PRIMARY KEY IDENTITY(1,1) NOT NULL,
Tipo VARCHAR(20) NOT NULL,
Direccion VARCHAR(45) NOT NULL,
Preferencia BIT NOT NULL,
);

/*Tabla Compuesta entre usuario y direccion*/
CREATE TABLE Direccion_Usuario(
Direccion_ID INT NOT NULL,
Usuario_ID INT NOT NULL,
PRIMARY KEY(Direccion_ID,Usuario_ID),

FOREIGN KEY (Usuario_ID) REFERENCES Usuario(Usuario_ID),
FOREIGN KEY (Direccion_ID) REFERENCES Direccion(Direccion_ID)
);

CREATE TABLE Pedido(
Pedido_ID INT PRIMARY KEY IDENTITY(1,1),
Usuario_ID INT NOT NULL,
Fecha_Pedido VARCHAR(50) NOT NULL,
Cantidad INT NOT NULL,
Estado VARCHAR(50) NOT NULL,

FOREIGN KEY (Usuario_ID) REFERENCES Usuario(Usuario_ID)
);
